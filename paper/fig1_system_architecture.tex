\begin{tikzpicture}[
    node distance=0.55cm and 0.45cm,

% --- Wearable sensing ---
\node[sensor] (unicorn) {Unicorn Hybrid Black\\8-ch EEG @ 250 Hz};

% --- Offline pipeline ---
\node[gui, below=0.9cm of collect, xshift=-1.8cm] (gui) {Unified GUI\\Collect $\rightarrow$ Preprocess $\rightarrow$ Train};

\node[model, right=of pre] (dl) {Deep Models\\EEGNet / CTNet / FBMSNet};

\node[box, right=of dl] (npy) {Subject-specific\\checkpoints (.pth)};

% --- Real-time path ---
\node[proc, below=0.9cm of pre, xshift=1.2cm] (rtpre) {Real-time Buffer\\1 s window, 0.5 s update};

\node[model, right=of rtpre] (infer) {Inference\\softmax class};

\node[act, right=of infer] (kinova) {Kinova Gen3 Arm\\left / right pose};

% --- Optional fusion ---
\node[box, below=0.55cm of infer] (cv) {OpenCV bar track\\(hybrid mode)};

% --- Arrows offline ---
\draw[arr] (unicorn) -- (collect);

\draw[arr] (dl) -- (npy);
\draw[arr] (gui.east) -- (pre.west);

% --- Arrows real-time ---

\draw[arr] (npy.south) |- (infer.north);
\draw[arr] (rtpre) -- (infer);
\draw[arr] (infer) -- (kinova);
\draw[arr] (cv.north) -- (infer.south);

% --- Labels ---
\node[font=\scriptsize, above=0.05cm of collect] {Phase-tagged CSV};
\node[font=\scriptsize, above=0.05cm of rtpre] {Streaming EEG};
\node[font=\scriptsize, below=0.02cm of cv] {match = fast / mismatch = slow};

\end{tikzpicture}%
}
\caption{End-to-end wearable EEG pipeline for motor-imagery decoding and Kinova robotic arm control. Offline training uses subject-specific models; real-time control applies 1\,s windows updated every 0.5\,s. Optional EEG--vision fusion modulates arm speed.}
